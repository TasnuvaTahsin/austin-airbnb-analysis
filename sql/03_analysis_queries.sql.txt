SELECT
    neighbourhood,
    COUNT(*) AS number_of_listings,
    ROUND(AVG(price), 2) AS avg_price,
    ROUND(MIN(price), 2) AS min_price,
    ROUND(MAX(price), 2) AS max_price
FROM listings_clean
WHERE price IS NOT NULL
GROUP BY neighbourhood
ORDER BY avg_price DESC;

-- Average price by room type
SELECT
    room_type,
    COUNT(*) AS number_of_listings,
    ROUND(AVG(price), 2) AS avg_price,
    ROUND(AVG(accommodates), 1) AS avg_accommodates,
    ROUND(AVG(bedrooms), 1) AS avg_bedrooms
FROM listings_clean
WHERE price IS NOT NULL
GROUP BY room_type
ORDER BY avg_price DESC;

-- Average price by neighborhood AND room type combined
SELECT
    neighbourhood,
    room_type,
    COUNT(*) AS number_of_listings,
    ROUND(AVG(price), 2) AS avg_price
FROM listings_clean
WHERE price IS NOT NULL
GROUP BY neighbourhood, room_type
HAVING COUNT(*) >= 5
ORDER BY neighbourhood, avg_price DESC;

-- Occupancy rate by neighborhood
SELECT
    l.neighbourhood,
    COUNT(*) AS total_nights,
    SUM(CASE WHEN c.available = false THEN 1 ELSE 0 END) AS booked_nights,
    ROUND(
        100.0 * SUM(CASE WHEN c.available = false THEN 1 ELSE 0 END) / COUNT(*),
        1
    ) AS occupancy_pct
FROM calendar_clean c
JOIN listings_clean l ON c.listing_id = l.listing_id
GROUP BY l.neighbourhood
ORDER BY occupancy_pct DESC;

-- Combined view: average price and occupancy % by neighborhood
SELECT
    l.neighbourhood,
    COUNT(DISTINCT l.listing_id) AS number_of_listings,
    ROUND(AVG(l.price), 2) AS avg_price,
    ROUND(
        100.0 * SUM(CASE WHEN c.available = false THEN 1 ELSE 0 END) / COUNT(*),
        1
    ) AS occupancy_pct
FROM listings_clean l
JOIN calendar_clean c ON l.listing_id = c.listing_id
WHERE l.price IS NOT NULL
GROUP BY l.neighbourhood
HAVING COUNT(DISTINCT l.listing_id) >= 10
ORDER BY occupancy_pct DESC;

-- Price by bedroom count, room type, and superhost status
SELECT
    bedrooms,
    room_type,
    host_is_superhost,
    COUNT(*) AS number_of_listings,
    ROUND(AVG(price), 2) AS avg_price,
    ROUND(AVG(review_scores_rating), 2) AS avg_rating
FROM listings_clean
WHERE price IS NOT NULL
    AND bedrooms IS NOT NULL
    AND bedrooms BETWEEN 0 AND 5
GROUP BY bedrooms, room_type, host_is_superhost
HAVING COUNT(*) >= 5
ORDER BY bedrooms, room_type, host_is_superhost;

-- Price and rating by review count bucket
SELECT
    CASE
        WHEN number_of_reviews = 0 THEN '0 reviews'
        WHEN number_of_reviews BETWEEN 1 AND 10 THEN '1-10'
        WHEN number_of_reviews BETWEEN 11 AND 50 THEN '11-50'
        WHEN number_of_reviews BETWEEN 51 AND 150 THEN '51-150'
        ELSE '150+'
    END AS review_bucket,
    COUNT(*) AS number_of_listings,
    ROUND(AVG(price), 2) AS avg_price,
    ROUND(AVG(review_scores_rating), 2) AS avg_rating
FROM listings_clean
WHERE price IS NOT NULL
GROUP BY review_bucket
ORDER BY MIN(number_of_reviews);
